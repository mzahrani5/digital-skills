/* يعمل الكود بطريقتين: داخل الجدول (الإضافات ← Apps Script)،
   أو مشروعًا مستقلًا من script.google.com — وفي هذه الحالة ينشئ الجدول بنفسه */
function ss_() {
  const props = PropertiesService.getScriptProperties();
  const id = props.getProperty('SS_ID');
  if (id) { try { return SpreadsheetApp.openById(id); } catch (e) {} }
  try { const a = SpreadsheetApp.getActiveSpreadsheet(); if (a) { props.setProperty('SS_ID', a.getId()); return a; } } catch (e) {}
  const created = SpreadsheetApp.create('إنجازات المهارات الرقمية - الصف الرابع');
  props.setProperty('SS_ID', created.getId());
  return created;
}

function doPost(e) {
  try {
    const q = JSON.parse(e.postData.contents);
    if (q.checkName) return out_(checkName_(ss_(), q.cls, q.name));
    if (q.visit) return out_(visit_(ss_(), q));
    if (q.retryCheck) return out_(retryCheck_(ss_(), q));
    if (q.sentCheck) return out_(sentCheck_(ss_(), q));
    if (q.retryReq) return out_(retryReq_(ss_(), q));
    if (q.checkSent) return out_(checkSent_(ss_(), q));
  } catch (err) {}
  // أي استعلام لم يُعالَج أعلاه لا يصل أبدًا إلى حفظ الإنجاز
  try { const q0 = JSON.parse(e.postData.contents); if (q0 && (q0.checkName || q0.visit || q0.retryCheck || q0.checkSent || q0.sentCheck || q0.retryReq)) return out_({ ok: false, status: 'error', message: 'تعذّر تنفيذ الطلب.' }); } catch (err) { return out_({ ok: false, status: 'error', message: 'طلب غير صالح' }); }
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const d = JSON.parse(e.postData.contents);
    const ss = ss_();
    // طلب تقرير ولي الأمر فقط (إرسال الطالب يحمل حقل report نصيًا مع رقم الدرس فلا يُخلط به)
    if (d.starOpt && d.code != null && !d.lessonId) return out_(starOpt_(ss, d.code, !!d.on));
    if (d.report && d.code != null && !d.lessonId && !d.exam) return out_(parentReport_(ss, d.code, d.consent));
    if (d.contact) return out_(contact_(ss, d));
    if (d.survey) return out_(survey_(ss, d));
    if (d.admin) return out_(admin_(ss, d));
    const m = matchName_(ss, d.cls, d.name);
    if (m.status === 'ok') d.name = m.name;
    else if (m.status !== 'noroster') return out_({ ok: false, status: 'unknown_name', message: 'الاسم غير موجود في كشف الفصل. اكتب اسمك كما في كشف المدرسة.' });
    if (d.exam) return out_(handleExam_(ss, d));
    const sh = respSheet_(ss);
    const name = norm_(d.name), cls = norm_(d.cls), id = String(d.lessonId || '');
    if (!name || !cls || !id) return out_({ ok: false, status: 'error', message: 'بيانات ناقصة' });
    if ((d.score === undefined || d.score === null || d.score === '') && !d.task && !d.quiz) return out_({ ok: false, status: 'error', message: 'لا يوجد إنجاز لإرساله' });
    if (closedIds_().indexOf(id) >= 0) return out_({ ok: false, status: 'closed', message: 'أغلق المعلم استقبال هذا المشروع.' });

    const vals = sh.getDataRange().getValues();
    for (let i = vals.length - 1; i >= 1; i--) {
      if (sameStudent_(vals[i][1], d.name) && clsKey_(vals[i][2]) === clsKey_(d.cls) && String(vals[i][3]) === id) {
        if (isFakeRow_(vals[i])) { if (sh.getLastRow() > 2) sh.deleteRow(i + 1); else sh.getRange(i + 1, 1, 1, sh.getMaxColumns()).clearContent(); continue; }
        return out_({ ok: false, status: 'duplicate', score: fixScoreCell_(sh, i + 1, vals[i]) });
      }
    }
    sh.appendRow([new Date(), d.name, d.cls, id, d.lesson, safeScore_(d.score, d.task, d.quiz), d.task, d.quiz, d.reflect, d.answers || '', d.report || '', d.late ? 'متأخر' : 'في الموعد', JSON.stringify(d.steps || [])]);
    try { formatResp_(sh); } catch (err) { console.error(err); audit_(ss, 'خطأ في تنسيق الاستجابات', String(err)); }
    try { updateGrid_(ss); } catch (err) { console.error(err); audit_(ss, 'خطأ في تحديث الكشوف', String(err)); }
    try { retryUsed_(ss, d.name, d.cls, id); } catch (err) { console.error(err); }
    try { saveToClassFolder_(ss, d); } catch (err) { console.error(err); }
    return out_({ ok: true, status: 'saved' });
  } catch (err) {
    return out_({ ok: false, status: 'error', message: String(err) });
  } finally {
    lock.releaseLock();
  }
}

    if (!o || !o.on) return out_({ ok: false, status: 'none' });
    if (!o.url) return out_({ ok: false, status: 'setup', name: o.name });
    return out_({ ok: true, teacher: { slug: o.slug, name: o.name, school: o.school, region: o.region, grades: o.grades, url: o.url } });
  }

  const ss = ss_();
  if (e && e.parameter && e.parameter.action === 'news') { const cl = closedIds_(); const live = l => { const m = String(l || '').match(/[?&]lesson=([^&#]+)/); if (!m) return true; const ids = decodeURIComponent(m[1]).split(','); return ids.some(x => cl.indexOf(x.trim()) < 0); };

  return [
    SpreadsheetApp.newConditionalFormatRule().whenNumberGreaterThanOrEqualTo(8).setBackground('#d9f2e3').setFontColor('#17844e').setBold(true).setRanges([range]).build(),
    SpreadsheetApp.newConditionalFormatRule().whenNumberBetween(5, 7.99).setBackground('#fdf1d6').setFontColor('#a86a00').setBold(true).setRanges([range]).build(),
    SpreadsheetApp.newConditionalFormatRule().whenNumberLessThan(5).setBackground('#fbe4e4').setFontColor('#c93b3b').setBold(true).setRanges([range]).build()

  return SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo('متأخر').setBackground('#fde8d4').setFontColor('#b45309').setBold(true).setRanges([range]).build();

function norm_(s) {
  return String(s == null ? '' : s)
    .replace(/[٠-٩]/g, c => '٠١٢٣٤٥٦٧٨٩'.indexOf(c))
    .replace(/[أإآ]/g, 'ا').replace(/ة/g, 'ه')
    .replace(/\s+/g, ' ').trim();
}

/* توحيد اسم الفصل: «٤/١» و«4/1» و«رابع 1» = 4-1 */
function clsKey_(c) {
  const s = String(c == null ? '' : c).replace(/[٠-٩]/g, x => '٠١٢٣٤٥٦٧٨٩'.indexOf(x));
  const m = s.match(/(\d+)\D+(\d+)/);
  if (m) return m[1] + '-' + m[2];
  const one = s.match(/(\d+)/);
  const g = /خامس/.test(s) ? '5' : /سادس/.test(s) ? '6' : '4';
  return one ? g + '-' + one[1] : s.replace(/[\/\\]/g, '-').trim() || 'بدون فصل';
}

function clsLabel_(c) {
  const k = clsKey_(c), m = k.match(/^([456])-(\d+)$/);
  return m ? ({ '4': 'رابع ', '5': 'خامس ', '6': 'سادس ' })[m[1]] + m[2] : k;
}

/* ترتيب الدروس حسب المنهج: t1-u2-3 ، t1-u2-p (مشروع) ، t2-u4-x (إثرائي) */
function order_(id) {
  const gm = String(id).match(/^g([56])-/), gw = gm ? Number(gm[1]) * 100000 : 0;
  const m = String(id).replace(/^g[56]-/, '').match(/^t(\d)-u(\d)-(\w+)$/);
  if (!m) return 999999;
  const last = m[3] === 'p' ? 9 : m[3] === 'x' ? 8 : Number(m[3]);
  return gw + Number(m[1]) * 1000 + Number(m[2]) * 10 + last;
}

function updateGrid_(ss) {
  const vals = respSheet_(ss).getDataRange().getValues().slice(1).filter(r => String(r[3]));
  const roster = rosterList_(ss);
  buildGrid_(ss, SHEET_GRID, vals, 0, roster);
  const keys = [...new Set(vals.map(r => clsKey_(r[2])).concat(roster.map(r => clsKey_(r.cls))))].sort();
  keys.forEach((k, i) => buildGrid_(ss, 'كشف ' + clsLabel_(k.replace('-', '/')), vals.filter(r => clsKey_(r[2]) === k), i + 1, roster.filter(r => clsKey_(r.cls) === k)));
  try { buildAnalysis_(ss, vals); } catch (e) { console.error(e); }
  try { buildBadges_(ss, vals); } catch (e) { console.error(e); }
}

function buildGrid_(ss, sheetName, vals, pos, roster) {
  const lessons = {}, students = {};
  (roster || []).forEach(r => { const key = clsKey_(r.cls) + '|' + norm_(r.name); if (!students[key]) students[key] = { name: r.name, cls: clsLabel_(r.cls), s: {} }; });
  vals.forEach(r => {
    const id = String(r[3]);
    if (!id) return;
    lessons[id] = String(r[4]).split(':').pop().trim() || id;
    const key = clsKey_(r[2]) + '|' + norm_(r[1]);
    if (!students[key]) students[key] = { name: String(r[1]).trim(), cls: clsLabel_(r[2]), s: {} };
    if (students[key].s[id] == null) students[key].s[id] = Number(r[5]);
  });
  const ids = Object.keys(lessons).sort((a, b) => order_(a) - order_(b));
  const list = Object.values(students).sort((a, b) => a.cls.localeCompare(b.cls) || a.name.localeCompare(b.name, 'ar'));

  let g = ss.getSheetByName(sheetName);
  if (!g) { g = ss.insertSheet(sheetName, pos); }
  g.setRightToLeft(true);
  g.getBandings().forEach(b => b.remove());
  g.clear();
  g.setConditionalFormatRules([]);
  const head = ['م', 'الطالب', 'الفصل'].concat(ids.map(id => lessons[id]), ['عدد الدروس', 'المتوسط']);
  const rows = list.map((st, i) => {
    const sc = ids.map(id => (st.s[id] == null ? 'لم يسلّم' : st.s[id]));
    const done = sc.filter(v => typeof v === 'number');
    const avg = done.length ? Math.round(done.reduce((a, b) => a + b, 0) / done.length * 10) / 10 : '';
    return [i + 1, st.name, st.cls].concat(sc, [done.length, avg]);
  });
  g.getRange(1, 1, 1, head.length).setValues([head]).setFontWeight('bold').setBackground(NAVY).setFontColor('#ffffff').setWrap(true)
    .setVerticalAlignment('middle').setHorizontalAlignment('center').setFontSize(10.5);
  g.setRowHeight(1, 52);
  g.setFrozenRows(1);
  g.setFrozenColumns(Math.min(3, head.length));
  g.setColumnWidth(1, 40);
  g.setColumnWidth(2, 200);
  g.setColumnWidth(3, 80);
  for (let c = 4; c <= head.length; c++) g.setColumnWidth(c, 95);
  g.getRange(1, head.length - 1, 1, 2).setBackground('#0f5c52');
  if (rows.length) {
    g.getRange(2, 1, rows.length, head.length).setValues(rows).setHorizontalAlignment('center');
    g.getRange(2, 2, rows.length, 1).setHorizontalAlignment('right');
    const bg = rows.map(r => r.slice(3).map((v, j) => {
      if (j === ids.length) return null;
      if (typeof v !== 'number') return j < ids.length ? '#fde8e8' : null;
      return v >= 8 ? '#e1f4e8' : v >= 5 ? '#fdf1d6' : '#fbe4e4';
    }));
    g.getRange(2, 4, rows.length, ids.length + 2).setBackgrounds(bg);
    g.getRange(2, 1, rows.length, head.length).setFontSize(11).setVerticalAlignment('middle');
    g.getRange(2, 2, rows.length, 1).setFontWeight('bold');
    g.getRange(2, head.length, rows.length, 1).setFontWeight('bold').setFontSize(12).setNumberFormat('0.0');
    g.setRowHeights(2, rows.length, 30);
    if (ids.length) g.getRange(2, 4, rows.length, ids.length).setFontColor('#14283d');
  }
  g.getRange(1, 1, Math.max(rows.length + 1, 1), head.length).setBorder(true, true, true, true, true, true, '#c9d4de', SpreadsheetApp.BorderStyle.SOLID);
}

/* مجلد لكل فصل، وفيه ملف لكل طالب يجمع تقارير إنجازه */
function folderIn_(parent, name) {
  const it = parent.getFoldersByName(name);
  return it.hasNext() ? it.next() : parent.createFolder(name);
}

/* ملفات الطالب في Drive: البحث فقط دون إنشاء مجلدات */
function studentDocs_(ss, cls, name) {
  const out = [];
  const file = DriveApp.getFileById(ss.getId()), parents = file.getParents();
  const base = parents.hasNext() ? parents.next() : DriveApp.getRootFolder();
  const r = base.getFoldersByName(ROOT_FOLDER); if (!r.hasNext()) return out;
  const cf = r.next().getFoldersByName(clsLabel_(cls)); if (!cf.hasNext()) return out;
  const folder = cf.next(), want = norm_(name);
  const it = folder.getFiles();
  while (it.hasNext()) { const f = it.next(); const nm = f.getName().split(' — ')[0]; if (norm_(nm) === want || sameStudent_(nm, name)) out.push(f); }
  return out;
}
function renameStudentDoc_(ss, cls, oldName, newName) {
  studentDocs_(ss, cls, oldName).forEach(f => f.setName(String(newName).replace(/\s+/g, ' ').trim() + ' — تقارير الإنجاز'));
}
/* حذف كل ما يخص الطالب عند حذفه من الكشف: إنجازاته، اختباره، طلباته، فرصه، محاولاته المؤرشفة، وملف تقاريره في Drive */
function purgeStudent_(ss, cls, name) {
  let rows = 0, files = 0;
  const wipe = (sh, ni, ci) => {
    if (!sh || sh.getLastRow() < 2) return;
    const v = sh.getRange(2, 1, sh.getLastRow() - 1, Math.max(ni, ci) + 1).getValues();
    for (let i = v.length - 1; i >= 0; i--) {
      if (sameStudent_(v[i][ni], name) && clsKey_(v[i][ci]) === clsKey_(cls)) {
        if (sh.getLastRow() > 2) sh.deleteRow(i + 2); else sh.getRange(i + 2, 1, 1, sh.getMaxColumns()).clearContent();
        rows++;
      }
    }
  };
  wipe(ss.getSheetByName(SHEET_RESP), 1, 2);
  wipe(ss.getSheetByName(SHEET_EXAM), 1, 2);
  wipe(ss.getSheetByName(SHEET_GRANT), 2, 1);
  wipe(ss.getSheetByName(SHEET_ARCH), 2, 3);
  wipe(ss.getSheetByName(SHEET_REQ), 2, 1);
  try { studentDocs_(ss, cls, name).forEach(f => { f.setTrashed(true); files++; }); } catch (err) { console.error(err); }
  return { rows, files };
}
function saveToClassFolder_(ss, d) {
  const file = DriveApp.getFileById(ss.getId());
  const parents = file.getParents();
  const base = parents.hasNext() ? parents.next() : DriveApp.getRootFolder();
  const root = folderIn_(base, ROOT_FOLDER);
  const clsFolder = folderIn_(root, clsLabel_(d.cls));
  const docName = String(d.name).replace(/\s+/g, ' ').trim() + ' — تقارير الإنجاز';
  const it = clsFolder.getFilesByName(docName);
  let doc;
  if (it.hasNext()) {
    doc = DocumentApp.openById(it.next().getId());
  } else {
    doc = DocumentApp.create(docName);
    DriveApp.getFileById(doc.getId()).moveTo(clsFolder);
    const b0 = doc.getBody();
    b0.clear();

    /* بطاقة التقرير المصوّرة كما رآها الطالب */
    const blob = Utilities.newBlob(Utilities.base64Decode(d.image), 'image/jpeg', String(d.name).trim() + ' - تقرير.jpg');

    p.setAlignment(DocumentApp.HorizontalAlignment.CENTER);

    const w = 430;
    img.setWidth(w).setHeight(Math.round(w * (Number(d.imgH) || 1600) / (Number(d.imgW) || 1080)));
  } else {

  }
  doc.saveAndClose();
}

const SHEET_ANALYSIS = 'تحليل الأخطاء';
function buildAnalysis_(ss, vals) {
  const L = {};
  const add = (id, title, score, max, steps) => {
    if (!id) return;
    const o = L[id] = L[id] || { title: title, n: 0, sum: 0, max: max, steps: {} };
    o.n++; o.sum += Number(score) / max;
    (steps || []).forEach(s => { const k = String(s.t); const e = o.steps[k] = o.steps[k] || { wrong: 0, total: 0 }; e.total++; if (!s.ok) e.wrong++; });
  };
  vals.forEach(r => { let st = []; try { st = JSON.parse(r[12] || '[]'); } catch (e) {} add(String(r[3]), String(r[4]).split(':').pop().trim(), r[5], 10, st); });
  const ex = ss.getSheetByName(SHEET_EXAM);
  if (ex && ex.getLastRow() > 1) ex.getRange(2, 1, ex.getLastRow() - 1, HEAD_EX.length).getValues().forEach(r => {
    let tasks = []; try { tasks = JSON.parse(r[11] || '[]'); } catch (e) {}
    tasks.forEach(tk => add(tk.id, tk.title, tk.ratio, 1, tk.steps));
  });
  let g = ss.getSheetByName(SHEET_ANALYSIS);
  if (!g) g = ss.insertSheet(SHEET_ANALYSIS);
  g.setRightToLeft(true); g.getBandings().forEach(b => b.remove()); g.clear(); g.setConditionalFormatRules([]);
  const head = ['الدرس', 'عدد المسلّمين', 'متوسط الإتقان', 'الخطوة', 'عدد من أخطأ', 'نسبة الخطأ'];
  const rows = [];
  Object.keys(L).sort((a, b) => order_(a) - order_(b)).forEach(id => {
    const o = L[id], steps = Object.keys(o.steps).map(k => [k, o.steps[k]]).sort((a, b) => b[1].wrong / b[1].total - a[1].wrong / a[1].total);
    if (!steps.length) steps.push(['(لا توجد خطوات مسجّلة)', { wrong: 0, total: 1 }]);
    steps.forEach((s, k) => rows.push([k ? '' : o.title, k ? '' : o.n, k ? '' : Math.round(o.sum / o.n * 100) / 100, s[0], s[1].wrong, s[1].total ? s[1].wrong / s[1].total : 0]));
  });
  g.getRange(1, 1, 1, head.length).setValues([head]).setFontWeight('bold').setFontColor('#ffffff').setBackground('#7c2d12').setHorizontalAlignment('center').setWrap(true);
  g.setFrozenRows(1); g.setRowHeight(1, 38);
  [200, 95, 105, 330, 95, 95].forEach((w, i) => g.setColumnWidth(i + 1, w));
  if (rows.length) {
    g.getRange(2, 1, rows.length, head.length).setValues(rows).setVerticalAlignment('middle');
    g.getRange(2, 1, rows.length, 1).setFontWeight('bold');
    g.getRange(2, 2, rows.length, 3).setHorizontalAlignment('center');
    g.getRange(2, 4, rows.length, 1).setHorizontalAlignment('right');
    g.getRange(2, 3, rows.length, 1).setNumberFormat('0%');
    g.getRange(2, 5, rows.length, 2).setHorizontalAlignment('center');
    g.getRange(2, 6, rows.length, 1).setNumberFormat('0%').setFontWeight('bold');
    const pr = g.getRange(2, 6, rows.length, 1);
    g.setConditionalFormatRules([
      SpreadsheetApp.newConditionalFormatRule().whenNumberGreaterThanOrEqualTo(0.5).setBackground('#fbe4e4').setFontColor('#c93b3b').setRanges([pr]).build(),
      SpreadsheetApp.newConditionalFormatRule().whenNumberBetween(0.25, 0.4999).setBackground('#fdf1d6').setFontColor('#a86a00').setRanges([pr]).build(),
      SpreadsheetApp.newConditionalFormatRule().whenNumberLessThan(0.25).setBackground('#e6f7ee').setFontColor('#17844e').setRanges([pr]).build()]);
    g.getRange(1, 1, rows.length + 1, head.length).setBorder(true, true, true, true, true, true, '#e5d5cb', SpreadsheetApp.BorderStyle.SOLID);
  }
}

function badgesFrom_(vals) {
  const S = {};
  vals.forEach(r => {
    const id = String(r[3]), m = id.match(/^(t\d-u\d)-(\w+)$/); if (!m || m[2] === 'x') return;
    const key = clsKey_(r[2]) + '|' + norm_(r[1]);
    const o = S[key] = S[key] || { name: String(r[1]).trim(), cls: clsLabel_(r[2]), u: {} };
    const u = o.u[m[1]] = o.u[m[1]] || {};
    if (u[id] == null) u[id] = Number(r[5]);
  });
  return Object.values(S).map(o => {
    const got = Object.keys(o.u).filter(k => UNITS[k] && Object.keys(o.u[k]).length >= UNITS[k][1] && Object.values(o.u[k]).reduce((a, b) => a + b, 0) / Object.keys(o.u[k]).length >= 8);
    return { name: o.name, cls: o.cls, units: got.sort((a, b) => order_(a + '-1') - order_(b + '-1')) };
  }).filter(o => o.units.length);
}
function buildBadges_(ss, vals) {
  const list = badgesFrom_(vals).sort((a, b) => a.cls.localeCompare(b.cls) || b.units.length - a.units.length || a.name.localeCompare(b.name, 'ar'));

  g.setFrozenRows(1); g.setRowHeight(1, 36);
  [200, 90, 95, 520].forEach((w, i) => g.setColumnWidth(i + 1, w));
  if (list.length) {
    const rows = list.map(o => [o.name, o.cls, o.units.length, o.units.map(k => '🏅 ' + UNITS[k][0]).join('   ')]);

   يكتب ولي الأمر الرمز في صفحة «تقرير ولي الأمر» على الموقع، فيصله تقرير ابنه فقط.
   شغّل makeParentCodes لإنشاء الرموز لكل الطلاب، ولإنشاء مستند بطاقات الرموز للطباعة. */
function parentUrl_() { let sl = ''; try { sl = (JSON.parse(PropertiesService.getScriptProperties().getProperty('PROFILE') || '{}') || {}).slug || ''; } catch (e) {} return 'https://m-zahrani.com/' + (sl ? '?t=' + sl : '') + '#report'; }
const CODE_ABC = 'ABCDEFGHJKMNPQRSTUVWXYZ23456789';
function normCode_(c) {
  return String(c == null ? '' : c).replace(/[٠-٩]/g, x => '٠١٢٣٤٥٦٧٨٩'.indexOf(x)).toUpperCase().replace(/[^A-Z0-9]/g, '');
}
function newCode_(used) {
  for (;;) {
    let c = ''; for (let i = 0; i < 8; i++) c += CODE_ABC[Math.floor(Math.random() * CODE_ABC.length)];
    if (!used.has(c)) { used.add(c); return c.slice(0, 4) + '-' + c.slice(4); }
  }
}
/* يضيف رمزًا لكل طالب ليس له رمز، ولا يغيّر الرموز الموجودة */
function ensureParentCodes_(ss) {
  const sh = rosterSheet_(ss), n = sh.getLastRow();
  if (n < 2) return 0;
  const rng = sh.getRange(2, 1, n - 1, 3), v = rng.getValues();
  const used = new Set(v.map(r => normCode_(r[2])).filter(Boolean));
  let added = 0;
  v.forEach(r => { if (String(r[0]).trim() && String(r[1]).trim() && !normCode_(r[2])) { r[2] = newCode_(used); added++; } });
  if (added) sh.getRange(2, 3, n - 1, 1).setValues(v.map(r => [r[2]]));
  return added;
}

/* تقرير طالب واحد فقط لمن يملك رمزه */
function parentReport_(ss, code, consent) {
  const cache = CacheService.getScriptCache();
  const fails = Number(cache.get('pfail') || 0);
  if (fails > 300) return { ok: false, status: 'busy', message: 'محاولات كثيرة. حاول بعد ١٠ دقائق.' };
  const c = normCode_(code);
  const sh = rosterSheet_(ss), n = sh.getLastRow();
  const all = c.length === 8 && n > 1 ? sh.getRange(2, 1, n - 1, 4).getValues() : [];
  const idx = all.findIndex(r => normCode_(r[2]) === c && String(r[1]).trim());
  const row = idx >= 0 ? all[idx] : null;
  if (!row) { cache.put('pfail', String(fails + 1), 600); Utilities.sleep(700); return { ok: false, status: 'invalid' }; }
  if (!row[3]) {
    if (!consent) return { ok: false, status: 'consent', message: 'يلزم موافقة ولي الأمر أولًا.' };
    sh.getRange(idx + 2, 4).setValue(new Date());
    audit_(ss, 'موافقة ولي أمر', String(row[1]) + ' — ' + String(row[0]));
  }
  const name = String(row[1]).replace(/\s+/g, ' ').trim(), ck = clsKey_(row[0]);
  const mine = r => norm_(r[1]) === norm_(name) && clsKey_(r[2]) === ck;
  const parse = x => { try { const v = JSON.parse(x || '[]'); return Array.isArray(v) ? v : []; } catch (e) { return []; } };
  const iso = d => d instanceof Date ? d.toISOString() : String(d || '');
  const resp = respSheet_(ss).getDataRange().getValues().slice(1).filter(r => r[3]);
  const classIds = {};
  resp.filter(r => clsKey_(r[2]) === ck).forEach(r => { if (!classIds[r[3]]) classIds[r[3]] = String(r[4]); });
  const lessons = resp.filter(mine).map(r => ({ id: String(r[3]), title: String(r[4]), at: iso(r[0]), score: Number(r[5]), task: String(r[6]), quiz: String(r[7]),
    answers: String(r[9]), late: String(r[11]) === 'متأخر', steps: parse(r[12]) })).sort((a, b) => order_(a.id) - order_(b.id));
  const got = new Set(lessons.map(l => l.id));
  const missing = Object.keys(classIds).filter(id => !got.has(id)).sort((a, b) => order_(a) - order_(b)).map(id => ({ id: id, title: classIds[id] }));
  const exSh = ss.getSheetByName(SHEET_EXAM);
  const exams = exSh ? exSh.getDataRange().getValues().slice(1).filter(r => r[3] && mine(r)).map(r => ({ at: iso(r[0]), lessons: String(r[4]), score: Number(r[5]), total: Number(r[6]),
    pct: Number(r[7]), tasks: String(r[8]), late: String(r[10]) === 'متأخر', steps: parse(r[11]) })) : [];
  const b = badgesFrom_(resp).find(o => norm_(o.name) === norm_(name) && clsKey_(o.cls) === ck);
  const badges = b ? b.units.map(u => {
    const sc = lessons.filter(l => l.id.indexOf(u + '-') === 0 && !/-x$/.test(l.id)).map(l => l.score);
    return { unit: u, title: UNITS[u] ? UNITS[u][0] : u, avg: sc.length ? Math.round(sc.reduce((x, y) => x + y, 0) / sc.length * 10) / 10 : 0 };
  }) : [];
  return { ok: true, student: { name: name, cls: clsLabel_(row[0]) }, stars: starsOkMap_()[c] !== -1, starsOn: starsOn_(), lessons: lessons, missing: missing, exams: exams, badges: badges, at: new Date().toISOString() };
}

/* ===================== كشف الدرجات والمتابعة (صفحة المعلم ← 📒 الكشف) =====================
   يُحفظ الكشف كاملًا في ملف واحد داخل Google Drive (ملف المعلم صاحب الجدول)، ولا يُقرأ أو يُكتب إلا برمز الإدارة.
   كل حفظ يحمل وقته (savedAt)؛ النسخة الأقدم لا تكتب فوق الأحدث حتى لو أرسلها جهاز آخر متأخرًا. */
const KASHF_NAME = 'كشف الدرجات والمتابعة — بيانات (لا تحذف هذا الملف).json';
function kashfFile_() {
  const props = PropertiesService.getScriptProperties();
  const id = props.getProperty('KASHF_FILE');
  if (id) { try { const f = DriveApp.getFileById(id); if (!f.isTrashed()) return f; } catch (e) {} }
  const f = DriveApp.createFile(KASHF_NAME, '{}', MimeType.PLAIN_TEXT);
  props.setProperty('KASHF_FILE', f.getId());
  return f;
}
function kashfRead_() { try { return JSON.parse(kashfFile_().getBlob().getDataAsString('UTF-8') || '{}') || {}; } catch (e) { return {}; } }
function kashfGet_(d) {
  const k = kashfRead_();
  const o = { ok: true, savedAt: Number(k.savedAt || 0), logoAt: Number(k.logoAt || 0) };
  if (o.savedAt > Number(d.since || 0)) o.data = k.data || null;
  if (o.logoAt > Number(d.logoSince || 0)) o.logo = k.logo || '';
  return o;
}
function kashfSet_(d) {
  const k = kashfRead_(); let changed = false;
  if (d.data != null) {
    const at = Number(d.savedAt || 0);
    if (typeof d.data !== 'string' || d.data.length > 3000000 || !at) return { ok: false, message: 'بيانات الكشف غير صالحة.' };
    if (at < Number(k.savedAt || 0)) return { ok: true, stale: true, savedAt: Number(k.savedAt || 0), logoAt: Number(k.logoAt || 0) };
    k.data = d.data; k.savedAt = at; changed = true;
  }
  if (d.logo != null) {
    const la = Number(d.logoAt || 0);
    if (la >= Number(k.logoAt || 0)) { k.logo = String(d.logo).slice(0, 600000); k.logoAt = la; changed = true; }
  }
  if (changed) kashfFile_().setContent(JSON.stringify(k));
  return { ok: true, savedAt: Number(k.savedAt || 0), logoAt: Number(k.logoAt || 0) };
}
/* ===================== منح فرصة أخرى =====================
   يحذف إرسال الطالب في درس (أو اختبار) فيستطيع حله وإرساله من جديد.
   وتتحقق صفحة الطالب عند فتحها: إذا لم يعد إرساله موجودًا في الجدول يُفتح الدرس له تلقائيًا. */
function sentOf_(ss, cls, name) {
  const mine = r => norm_(r[1]) === norm_(name) && clsKey_(r[2]) === clsKey_(cls);
  const iso = v => v instanceof Date ? v.toISOString() : String(v || '');
  const out = respSheet_(ss).getDataRange().getValues().slice(1).filter(r => r[3] && mine(r))
    .map(r => ({ kind: 'lesson', id: String(r[3]), title: String(r[4]).split('·').pop().trim(), score: Number(r[5]), of: 10, at: iso(r[0]) }))
    .sort((a, b) => order_(a.id) - order_(b.id));
  const ex = ss.getSheetByName(SHEET_EXAM);
  if (ex && ex.getLastRow() > 1) ex.getDataRange().getValues().slice(1).filter(r => r[3] && mine(r))
    .forEach(r => out.push({ kind: 'exam', id: String(r[3]), title: 'الاختبار النهائي', score: Number(r[5]), of: Number(r[6]) || 20, at: iso(r[0]) }));
  return out;
}
function retry_(ss, cls, name, id, exam) {
  const sh = exam ? ss.getSheetByName(SHEET_EXAM) : respSheet_(ss);
  if (!sh || sh.getLastRow() < 2 || !id) return 0;
  const v = sh.getRange(2, 1, sh.getLastRow() - 1, 4).getValues();
  let k = 0;
  for (let i = v.length - 1; i >= 0; i--) {
    if (norm_(v[i][1]) === norm_(name) && clsKey_(v[i][2]) === clsKey_(cls) && String(v[i][3]) === id) {
      if (sh.getLastRow() > 2) sh.deleteRow(i + 2); else sh.getRange(i + 2, 1, 1, sh.getMaxColumns()).clearContent();
      k++;
    }
  }
  if (k) { audit_(ss, 'فرصة أخرى', String(name) + ' — ' + String(cls) + ' — ' + id); try { updateGrid_(ss); } catch (e) { console.error(e); } }
  return k;
}

  const ids = (Array.isArray(q.ids) ? q.ids : []).map(String).slice(0, 80);

  const mine = r => norm_(r[1]) === norm_(name) && clsKey_(r[2]) === clsKey_(cls);

function purgeMessages_(ss) {
  const sh = ss.getSheetByName(SHEET_MSG);
  if (!sh || sh.getLastRow() < 2) return 0;
  const now = Date.now(), day = 864e5, v = sh.getRange(2, 1, sh.getLastRow() - 1, 6).getValues();
  let k = 0;
  for (let i = v.length - 1; i >= 0; i--) {
    const at = v[i][0] instanceof Date ? v[i][0].getTime() : 0, age = at ? (now - at) / day : 0;
    if (at && (age > MSG_KEEP_MAX_DAYS || (String(v[i][5]) === 'تم الرد' && age > MSG_KEEP_DONE_DAYS))) {
      if (sh.getLastRow() > 2) sh.deleteRow(i + 2); else sh.getRange(i + 2, 1, 1, 6).clearContent();
      k++;
    }
  }
  if (k) audit_(ss, 'حذف تلقائي للرسائل', k + ' رسالة تجاوزت مدة الاحتفاظ');
  return k;
}

/* درجة سليمة دائمًا: إن وصلت غير رقمية تُحسب من المهمة (س/٧) والأسئلة (س من ن) */
function safeScore_(score, task, quiz) {
  const n = Number(score);
  if (isFinite(n) && String(score) !== '' && score !== null) return Math.max(0, Math.min(10, n));
  const toN = t => Number(String(t || '').replace(/[٠-٩]/g, x => '٠١٢٣٤٥٦٧٨٩'.indexOf(x)));
  const tm = String(task || '').replace(/[٠-٩]/g, x => '٠١٢٣٤٥٦٧٨٩'.indexOf(x)).match(/\((\d+)\s*\/\s*7\)/);
  const qm = String(quiz || '').replace(/[٠-٩]/g, x => '٠١٢٣٤٥٦٧٨٩'.indexOf(x)).match(/^(\d+)/);
  const v = (tm ? toN(tm[1]) : 0) + (qm ? toN(qm[1]) : 0);
  return isFinite(v) ? Math.max(0, Math.min(10, v)) : 0;
}

  if (!title && !(kind === 'bar' && text)) return null;
  const pts = (Array.isArray(d.points) ? d.points : String(d.points || '').split('\n')).map(x => t(x, 80)).filter(Boolean).slice(0, 3).join('\n');
  const link = /^https?:\/\/|^#/.test(String(d.link || '').trim()) ? String(d.link).trim().slice(0, 300) : '';
  const dt = v => /^\d{4}-\d{2}-\d{2}$/.test(String(v || '')) ? String(v) : '';
  return [new Date(), kind, PROMO_TONES.indexOf(d.tone) >= 0 ? d.tone : 'blue', t(d.badge, 20), title, text, pts, t(d.btn, 30), link, dt(d.start), dt(d.end), d.countdown ? 'نعم' : 'لا', 'مفعّل', PROMO_ICONS.indexOf(d.icon) >= 0 ? d.icon : 'bell'];
}
function promosList_(ss, all) {
  const sh = ss.getSheetByName(SHEET_PROMO);
  if (!sh || sh.getLastRow() < 2) return [];
  const today = ymd_(new Date());
  return sh.getRange(2, 1, sh.getLastRow() - 1, 14).getValues().map((r, i) => {
    const kind = PROMO_KINDS[String(r[1]).trim()] ? String(r[1]).trim() : (Object.keys(PROMO_KINDS).filter(k => PROMO_KINDS[k] === String(r[1]).trim())[0] || 'card');
    return { row: i + 2, id: r[0] instanceof Date ? String(r[0].getTime()) : String(i + 2), kind: kind, tone: String(r[2] || '').trim(), badge: String(r[3] || ''), title: String(r[4] || ''), text: String(r[5] || ''),
      points: String(r[6] || '').split('\n').map(x => x.trim()).filter(Boolean).slice(0, 3), btn: String(r[7] || ''), link: String(r[8] || ''),
      start: ymd_(r[9]), end: ymd_(r[10]), countdown: /^(نعم|yes|true)$/i.test(String(r[11]).trim()), active: String(r[12]).trim() !== 'موقوف', icon: PROMO_ICONS.indexOf(String(r[13]).trim()) >= 0 ? String(r[13]).trim() : 'bell' };
  }).filter(p => (p.title || p.text) && (all || (p.active && (!p.start || p.start <= today) && (!p.end || p.end >= today))))
    .reverse().slice(0, all ? 50 : 6).map(p => { if (!all) { delete p.row; delete p.active; } return p; });
}


/* ===================== التحقق من اسم الطالب دون كشف قائمة الأسماء ===================== */
function nameTokens_(s) { return norm_(s).replace(/ى/g, 'ي').replace(/ؤ|ئ/g, 'ء').replace(/(^|\s)(عبد|ابو)\s+/g, '$1$2').replace(/[^؀-ۿa-zA-Z0-9 ]/g, ' ').replace(/\s+/g, ' ').trim().split(' ').filter(w => w && w !== 'بن' && w !== 'ابن'); }
function matchIn_(list, name) {
  const T = nameTokens_(name);
  if (T.length < 2) return { status: 'short' };
  const exact = list.filter(r => nameTokens_(r.name).join(' ') === T.join(' '));
  if (exact.length === 1) return { status: 'ok', rows: [exact[0]] };
  const sub = list.filter(r => {
    const R0 = nameTokens_(r.name);
    if (R0[0] !== T[0] || R0[R0.length - 1] !== T[T.length - 1]) return false;
    let i = 0; for (const w of R0) { if (w === T[i]) i++; } return i === T.length;
  });
  if (sub.length === 1) return { status: 'ok', rows: [sub[0]] };
  if (sub.length > 1 || exact.length > 1) return { status: 'ambiguous', rows: (exact.length > 1 ? exact : sub).slice(0, 5) };
  // تقريب الأخطاء الإملائية: الاسم الأول والعائلة متقاربان، وباقي الكلمات متقاربة بالترتيب
  const near = (a, b) => a === b || lev_(a, b) <= (Math.max(a.length, b.length) >= 6 ? 2 : 1);
  const scored = list.map(r => {
    const R0 = nameTokens_(r.name);
    if (!near(R0[0], T[0]) || !near(R0[R0.length - 1], T[T.length - 1])) return null;
    let i = 0, cost = 0;
    for (const w of R0) { if (i < T.length && near(w, T[i])) { cost += lev_(w, T[i]); i++; } }
    return i === T.length ? { r: r, cost: cost } : null;
  }).filter(Boolean).sort((a, b) => a.cost - b.cost).slice(0, 3);
  if (scored.length) return { status: 'suggest', rows: scored.map(x => x.r) };
  // أسماء بالإنجليزية في الكشف والطالب كتب بالعربية (أو العكس): مقارنة بالنطق
  const S = skel_(name);
  if (S.length >= 2) {
    const nearS = (a, b) => a === b || (Math.min(a.length, b.length) >= 3 && lev_(a, b) <= 1);
    const cross = list.map(r => {
      const R0 = skel_(r.name);
      if (R0.length < 2 || !nearS(R0[0], S[0]) || !nearS(R0[R0.length - 1], S[S.length - 1])) return null;
      let i = 0, cost = 0;
      for (const w of R0) { if (i < S.length && nearS(w, S[i])) { cost += lev_(w, S[i]); i++; } }
      return i === S.length ? { r: r, cost: cost } : null;
    }).filter(Boolean).sort((a, b) => a.cost - b.cost).slice(0, 3);
    if (cross.length) return { status: 'suggest', rows: cross.map(x => x.r) };
  }
  return { status: 'notfound' };
}
function matchName_(ss, cls, name) {
  const list = rosterList_(ss).filter(r => clsKey_(r.cls) === clsKey_(cls));
  if (!list.length) return { status: 'noroster' };
  const m = matchIn_(list, name);
  if (m.status === 'ok') return { status: 'ok', name: m.rows[0].name };
  return m.rows ? { status: m.status, options: m.rows.map(r => r.name) } : m;
}
function checkName_(ss, cls, name) {
  const cache = CacheService.getScriptCache();
  const fails = Number(cache.get('nfail') || 0);
  if (fails > 400) return { ok: false, status: 'busy' };
  if (!String(cls || '').trim()) {
    // البحث في كل الفصول وتحديد فصل الطالب تلقائيًا
    const list = rosterList_(ss);
    if (!list.length) return { ok: true, noroster: true };
    const T = nameTokens_(name);
    if (T.length < 2) return { ok: false, status: 'short' };
    const m = matchIn_(list, name);
    if (m.status === 'ok') return { ok: true, name: m.rows[0].name, cls: m.rows[0].cls };
    if (m.status === 'ambiguous' && T.length < 3) return { ok: false, status: 'need3' };
    if (m.rows) return { ok: false, status: m.status, options: m.rows.map(r => ({ name: r.name, cls: r.cls })) };
    cache.put('nfail', String(fails + 1), 600); Utilities.sleep(400);
    return { ok: false, status: m.status };
  }
  const m = matchName_(ss, cls, name);
  if (m.status === 'ok') return { ok: true, name: m.name };
  if (m.status === 'noroster') return { ok: true, noroster: true };
  if (m.status === 'notfound') { cache.put('nfail', String(fails + 1), 600); Utilities.sleep(400); }
  return m.status === 'ambiguous' || m.status === 'suggest' ? { ok: false, status: m.status, options: m.options } : { ok: false, status: m.status };
}

function lev_(a, b) {
  if (a === b) return 0;
  const m = a.length, n = b.length; if (!m || !n) return m || n;
  let prev = []; for (let j = 0; j <= n; j++) prev[j] = j;
  for (let i = 1; i <= m; i++) {
    const cur = [i];
    for (let j = 1; j <= n; j++) cur[j] = Math.min(prev[j] + 1, cur[j - 1] + 1, prev[j - 1] + (a[i - 1] === b[j - 1] ? 0 : 1));
    prev = cur;
  }
  return prev[n];
}

/* هيكل نطق الاسم: يحوّل العربي والإنجليزي إلى حروف ساكنة متقاربة لمقارنة «محمد» بـ Mohammed */
function skel_(s) {
  const AR = { 'ب':'b','ت':'t','ط':'t','ث':'t','ج':'j','ح':'h','ه':'h','ة':'h','خ':'k','د':'d','ض':'d','ذ':'z','ز':'z','ظ':'z','ر':'r','س':'s','ص':'s','ش':'x','غ':'g','ف':'f','ق':'k','ك':'k','ل':'l','م':'m','ن':'n','و':'w','ي':'y','ى':'','ا':'a','أ':'a','إ':'a','آ':'a','ع':'','ء':'','ئ':'','ؤ':'' };
  return String(s || '').replace(/[ًٌٍَُِّْـ]/g, '').replace(/(^|\s)(عبد|ابو|أبو)\s+/g, '$1$2').replace(/\bAbd\s+/gi, 'Abd').replace(/-/g, '').split(/\s+/).filter(w => w && !/^(بن|ابن|bin|ibn)$/i.test(w)).map(w => {
    let x;
    if (/[؀-ۿ]/.test(w)) {
      x = w.split('').map((c, i) => { const v = AR[c]; if (v === undefined) return ''; if ((c === 'و' || c === 'ي' || /[اأإآ]/.test(c)) && i > 0) return ''; return v; }).join('');
    } else {
      x = w.toLowerCase().replace(/[^a-z]/g, '').replace(/ph/g, 'f').replace(/sh/g, 'x').replace(/kh/g, 'k').replace(/gh/g, 'g').replace(/th/g, 't').replace(/dh/g, 'z').replace(/q/g, 'k').replace(/c/g, 'k').replace(/e/g, 'a').replace(/o/g, 'a').replace(/u/g, 'a').replace(/i/g, 'a');
      x = x.split('').map((c, i) => (i > 0 && /[aiwy]/.test(c)) ? '' : c).join('');
    }
    x = x.replace(/^a/, '').replace(/(.)\1+/g, '$1');
    return x;
  }).filter(Boolean);
}

const SHEET_VRAW = 'زيارات (مؤقت)', SHEET_VDAY = 'إحصاءات الزيارات', SHEET_VPAGE = 'الصفحات الأكثر زيارة';
function visit_(ss, q) {
  const cache = CacheService.getScriptCache(), k = 'v' + Math.floor(Date.now() / 60000), n = Number(cache.get(k) || 0);
  if (n > 400) return { ok: true };
  cache.put(k, String(n + 1), 120);
  let sh = ss.getSheetByName(SHEET_VRAW);
  if (!sh) { sh = ss.insertSheet(SHEET_VRAW); sh.appendRow(['التاريخ', 'الصفحة', 'الجهاز', 'المصدر', 'زيارة جديدة', 'زائر اليوم', 'زائر جديد', 'زائر الأسبوع', 'زائر الشهر']); sh.setRightToLeft(true); }
  const tz = Session.getScriptTimeZone() || 'Asia/Riyadh';
  sh.appendRow([Utilities.formatDate(new Date(), tz, 'yyyy-MM-dd'), String(q.p || 'غير معروف').slice(0, 60), q.d === 'm' ? 'جوال' : 'حاسب', q.s === 'proj' ? 'المنصة' : 'الموقع', q.n ? 1 : 0, q.u ? 1 : 0, q.f ? 1 : 0, q.w ? 1 : 0, q.mo ? 1 : 0]);
  if (sh.getLastRow() > 1500) { const lock = LockService.getScriptLock(); if (lock.tryLock(3000)) { try { rollVisits_(ss); } finally { lock.releaseLock(); } } }
  return { ok: true };
}

    const o = D[d] = D[d] || [0, 0, 0, 0, 0, 0, 0, 0, 0];
    o[1]++; if (r[4]) o[0]++; if (r[2] === 'جوال') o[2]++; else o[3]++; if (r[3] === 'المنصة') o[4]++; if (Number(r[5]) === 1) o[5]++; if (Number(r[6]) === 1) o[6]++; if (Number(r[7]) === 1) o[7]++; if (Number(r[8]) === 1) o[8]++;

  });
  if (day.getLastColumn() < 10) day.getRange(1, 7, 1, 4).setValues([['زوار اليوم (بلا تكرار)', 'زوار جدد', 'زوار الأسبوع', 'زوار الشهر']]);
  const dv = day.getLastRow() > 1 ? day.getRange(2, 1, day.getLastRow() - 1, 10).getValues() : [];

  if (dv.length) day.getRange(2, 1, dv.length, 10).setValues(dv.map(r => [String(r[0]).replace(/^'/, '').length === 10 ? "'" + String(r[0]).replace(/^'/, '') : r[0], r[1], r[2], r[3], r[4], r[5], Number(r[6] || 0), Number(r[7] || 0), Number(r[8] || 0), Number(r[9] || 0)]));
  const pv = pg.getLastRow() > 1 ? pg.getRange(2, 1, pg.getLastRow() - 1, 3).getValues() : [];

function stats_(ss) {
  rollVisits_(ss);
  const tz = Session.getScriptTimeZone() || 'Asia/Riyadh', day = ss.getSheetByName(SHEET_VDAY), pg = ss.getSheetByName(SHEET_VPAGE);
  const ds = d => d instanceof Date ? Utilities.formatDate(d, tz, 'yyyy-MM-dd') : String(d).replace(/^'/, '');
  const days = day && day.getLastRow() > 1 ? day.getRange(2, 1, day.getLastRow() - 1, Math.max(6, Math.min(10, day.getLastColumn()))).getValues().map(r => ({ d: ds(r[0]), s: +r[1] || 0, v: +r[2] || 0, m: +r[3] || 0, c: +r[4] || 0, p: +r[5] || 0, u: +r[6] || 0, f: +r[7] || 0, w: +r[8] || 0, mo: +r[9] || 0 })) : [];
  const since = Utilities.formatDate(new Date(Date.now() - 29 * 864e5), tz, 'yyyy-MM-dd');
  const top = {};
  if (pg && pg.getLastRow() > 1) pg.getRange(2, 1, pg.getLastRow() - 1, 3).getValues().forEach(r => { if (ds(r[0]) >= since) top[r[1]] = (top[r[1]] || 0) + (+r[2] || 0); });
  return { today: Utilities.formatDate(new Date(), tz, 'yyyy-MM-dd'), days: days.filter(x => x.d >= Utilities.formatDate(new Date(Date.now() - 89 * 864e5), tz, 'yyyy-MM-dd')).sort((a, b) => a.d < b.d ? -1 : 1),
    total: days.reduce((a, x) => ({ s: a.s + x.s, v: a.v + x.v, f: a.f + x.f }), { s: 0, v: 0, f: 0 }),
    top: Object.keys(top).map(k => [k, top[k]]).sort((a, b) => b[1] - a[1]).slice(0, 12) };
}
